class TransactionModel {
  final String id;
  final String userId;
  final String type; // 'payment', 'refund', 'ad_payment', 'withdrawal'
  final double amount;
  final String currency;
  final String paymentMethod; // 'cash', 'wallet', 'card'
  final String status; // 'completed', 'pending', 'failed', 'cancelled'
  final String description;
  final String? orderId;
  final String? adId;
  final double balanceBefore;
  final double balanceAfter;
  final DateTime createdAt;

  TransactionModel({
    required this.id,
    required this.userId,
    required this.type,
    required this.amount,
    this.currency = 'JOD',
    required this.paymentMethod,
    required this.status,
    required this.description,
    this.orderId,
    this.adId,
    required this.balanceBefore,
    required this.balanceAfter,
    required this.createdAt,
  });

  Map<String, dynamic> toMap() {
    return {
      'id': id,
      'userId': userId,
      'type': type,
      'amount': amount,
      'currency': currency,
      'paymentMethod': paymentMethod,
      'status': status,
      'description': description,
      'orderId': orderId,
      'adId': adId,
      'balanceBefore': balanceBefore,
      'balanceAfter': balanceAfter,
      'createdAt': createdAt.toIso8601String(),
    };
  }

  factory TransactionModel.fromMap(Map<String, dynamic> map) {
    return TransactionModel(
      id: map['id'] ?? '',
      userId: map['userId'] ?? '',
      type: map['type'] ?? 'payment',
      amount: (map['amount'] ?? 0).toDouble(),
      currency: map['currency'] ?? 'JOD',
      paymentMethod: map['paymentMethod'] ?? 'wallet',
      status: map['status'] ?? 'pending',
      description: map['description'] ?? '',
      orderId: map['orderId'],
      adId: map['adId'],
      balanceBefore: (map['balanceBefore'] ?? 0).toDouble(),
      balanceAfter: (map['balanceAfter'] ?? 0).toDouble(),
      createdAt: DateTime.parse(map['createdAt'] ?? DateTime.now().toIso8601String()),
    );
  }
}
